import os
os.environ['ONEAPI_DEVICE_SELECTOR'] = 'level_zero:gpu'
import xgboost as xgb
from sklearn.datasets import make_classification
import time
import json
def bench(label, params, X, y, rounds=100):
    dtrain = xgb.DMatrix(X, label=y)
    t0 = time.time()
    model = xgb.train(params, dtrain, num_boost_round=rounds, verbose_eval=False)
    t1 = time.time()
    cfg = json.loads(model.save_config())
    device = cfg['learner']['generic_param']['device']
    updater = cfg['learner']['gradient_booster']['gbtree_train_param']['updater_seq']
    print(f'{label}: {t1-t0:.3f}s  device={device} updater={updater}')
    return t1 - t0
# Binary classification (single class)
for n in [5000, 20000, 50000, 100000]:
    n_feat = 50
    X, y = make_classification(n_samples=n, n_features=n_feat, n_classes=2, n_informative=int(n_feat*0.6), random_state=42)
    by = (y == 0).astype(int)
    for device in ['cpu', 'sycl:gpu:0']:
        p = {'device': device, 'tree_method': 'hist', 'max_depth': 8, 'verbosity': 0}
        tag = f'n={n:>6d} feat={n_feat} {device:>12s}'
        bench(tag, p, X, by, rounds=200)
